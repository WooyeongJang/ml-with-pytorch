class ConvNet(nn.Module):
  def __init__(self):
    super().__init__()
    # 블록1: 28×28 → 14×14
    self.conv1 = nn.Conv2d(1,32,3,padding=1)
    self.bn1   = nn.BatchNorm2d(32)
    self.pool  = nn.MaxPool2d(2,2)
    # 블록2: 14×14 → 7×7
    self.conv2 = nn.Conv2d(32,64,3,padding=1)
    self.bn2   = nn.BatchNorm2d(64)
    # 분류 헤드
    self.fc1   = nn.Linear(64*7*7,128)
    self.drop  = nn.Dropout(0.5)
    self.fc2   = nn.Linear(128,10)
  def forward(self, x):
    x = self.pool(F.relu(self.bn1(self.conv1(x))))
    x = self.pool(F.relu(self.bn2(self.conv2(x))))
    x = x.view(-1, 64*7*7)
    return self.fc2(self.drop(F.relu(self.fc1(x))))

class SmileCNN(nn.Module):
  def __init__(self):
    super().__init__()
    self.features = nn.Sequential(
      nn.Conv2d(3,32,3,padding=1),
      nn.BatchNorm2d(32),
      nn.ReLU(),
      nn.MaxPool2d(2),
      nn.Conv2d(32,64,3,padding=1),
      nn.BatchNorm2d(64),
      nn.ReLU(),
      nn.MaxPool2d(2),

훈련 루프
for epoch in range(num_epochs):
  model.train()
  for imgs, labels in train_loader:
    optimizer.zero_grad()
    outputs = model(imgs)
    loss = criterion(outputs, labels)
    loss.backward()
    optimizer.step()
  model.eval()
  with torch.no_grad():
    correct = (model(val_imgs)
      .argmax(1) == val_labels).sum()
    acc = correct / len(val_labels)
  print(f'Epoch {epoch}: acc={acc:.4f}')

class SentimentLSTM(nn.Module):
  def __init__(self, vocab, emb, hid):
    super().__init__()
    self.emb  = nn.Embedding(vocab, emb,
                  padding_idx=1)
    self.lstm = nn.LSTM(
      emb, hid,
      num_layers=2,
      batch_first=True,
      dropout=0.5)
    self.fc = nn.Linear(hid, 1)
  def forward(self, x):
    emb = self.emb(x)
    out, (h, c) = self.lstm(emb)
    return self.fc(h[-1]).squeeze()

class CharRNN(nn.Module):
  def __init__(self, n_chars, emb, hid):
    super().__init__()
    self.emb  = nn.Embedding(n_chars, emb)
    self.lstm = nn.LSTM(emb, hid,
                  batch_first=True)
    self.fc   = nn.Linear(hid, n_chars)

def sample(model, seed, length,
           temperature=1.0):
  chars = list(seed)
  for _ in range(length):
    x = encode(chars[-seq_len:])
    logits = model(x)
    probs = softmax(logits/temperature)
    chars.append(sample_from(probs))
  return ''.join(chars)
temperature 파라미터 효과
낮은 온도 (0.5)
예측 가능, 반복적 텍스트. 가장 높은 확률 단어 선택.
중간 온도 (1.0)
균형 잡힌 생성. 기본값.
높은 온도 (1.5+)
창의적이지만 때로 무작위. 분포를 평탄화.
